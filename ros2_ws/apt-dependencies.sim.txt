# Simulation-only dependencies for the innate-os ROS2 workspace.
# Installed on top of apt-dependencies.common.txt for the published simulation
# image (see sim/Dockerfile and DEPENDENCIES_GUIDE.md). The robot does not use
# this file; it gets these packages transitively from the meta-packages in
# apt-dependencies.hardware.txt (navigation2, moveit, ...).
# Install with: xargs -a apt-dependencies.sim.txt sudo apt-get install -y

# Build tools used by the local sim launcher
openssh-client
pkg-config

# C++ libraries used by workspace packages (named explicitly because the sim
# build omits the meta-packages that would otherwise pull them in)
libeigen3-dev
libfcl-dev
libopencv-dev
libopenmpi-dev
libyaml-cpp-dev

# ROS 2 build/runtime basics
ros-humble-rclcpp-lifecycle
ros-humble-message-filters

# ROS 2 messages used by the sim stack
ros-humble-moveit-msgs
ros-humble-shape-msgs
ros-humble-stereo-msgs
ros-humble-tf2-msgs
ros-humble-trajectory-msgs
ros-humble-visualization-msgs

# Trimmed navigation stack (sim assembles nav from sub-packages instead of the
# ros-humble-navigation2 meta-package used on the robot)

# deactivate SLAM in navigation mode or it clobbers /map.
ros-humble-slam-toolbox-humble-lifecycle
